Pénzügyi adatok vizuális szűrése (Google Sheets)

Cél: Egy Google Táblázat adatainak gyors, vizuális feldolgozása. A szkript optimalizált módon (tömbök használatával) átvizsgálja a táblázat összes adatát, és a negatív értékeket tartalmazó cellákat automatikusan piros formázással látja el.

Nyelv: Google Apps Script (JavaScript)

function onOpen() {
  SpreadsheetApp.getUi().createMenu('Táblázat Szkriptek')
      .addItem('Negatív számok kiemelése', 'highlightNegatives')
      .addToUi();
}

function highlightNegatives() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var range = sheet.getDataRange();
  
  var values = range.getValues();
  var bgColors = range.getBackgrounds();
  var fontColors = range.getFontColors();
  var count = 0;

  for (var r = 0; r < values.length; r++) {
    for (var c = 0; c < values[r].length; c++) {
      if (typeof values[r][c] === 'number' && values[r][c] < 0) {
        bgColors[r][c] = '#ffcccc';
        fontColors[r][c] = '#cc0000';
        count++;
      }
    }
  }

  range.setBackgrounds(bgColors);
  range.setFontColors(fontColors);

  SpreadsheetApp.getUi().alert('Kész! ' + count + ' negatív értéket emeltem ki.');
}
